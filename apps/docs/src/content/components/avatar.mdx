
## Usage

```tsx
import { Avatar } from 'nyxis-ui';
// or, for stricter tree-shaking:
// import { Avatar } from 'nyxis-ui/avatar';

<Avatar>
  <AvatarImage src="/julio.png" alt="@juliodaal" />
  <AvatarFallback>JD</AvatarFallback>
</Avatar>;
```
