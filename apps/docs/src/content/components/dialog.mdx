## Installation

```bash
pnpm add nyxis-ui
```

## Usage

```tsx
import { Dialog } from 'nyxis-ui';
// or, for stricter tree-shaking:
// import { Dialog } from 'nyxis-ui/dialog';

<Dialog>
  <DialogTrigger asChild>
    <Button>Open</Button>
  </DialogTrigger>
  <DialogContent>
    <DialogHeader>
      <DialogTitle>Confirm</DialogTitle>
      <DialogDescription>Are you sure?</DialogDescription>
    </DialogHeader>
  </DialogContent>
</Dialog>;
```
