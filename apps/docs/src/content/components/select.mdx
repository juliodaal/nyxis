## Installation

```bash
pnpm add nyxis-ui
```

## Usage

```tsx
import { Select } from 'nyxis-ui';
// or, for stricter tree-shaking:
// import { Select } from 'nyxis-ui/select';

<Select>
  <SelectTrigger>
    <SelectValue placeholder="Pick one" />
  </SelectTrigger>
  <SelectContent>
    <SelectItem value="datev">DATEV</SelectItem>
    <SelectItem value="netsuite">NetSuite</SelectItem>
  </SelectContent>
</Select>;
```
