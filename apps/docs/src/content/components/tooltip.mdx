## Installation

```bash
pnpm add nyxis-ui
```

## Usage

```tsx
import { Tooltip } from 'nyxis-ui';
// or, for stricter tree-shaking:
// import { Tooltip } from 'nyxis-ui/tooltip';

<TooltipProvider>
  <Tooltip>
    <TooltipTrigger asChild>
      <Button>Hover</Button>
    </TooltipTrigger>
    <TooltipContent>92% confidence</TooltipContent>
  </Tooltip>
</TooltipProvider>;
```
