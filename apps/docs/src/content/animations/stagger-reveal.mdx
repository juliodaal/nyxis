## Installation

```bash
pnpm add nyxis-ui
```

## Usage

```tsx
import { StaggerReveal } from 'nyxis-ui';
// or, for stricter tree-shaking:
// import { StaggerReveal } from 'nyxis-ui/stagger-reveal';

<StaggerReveal>
  {items.map((item) => (
    <div key={item.id}>{item.label}</div>
  ))}
</StaggerReveal>;
```
